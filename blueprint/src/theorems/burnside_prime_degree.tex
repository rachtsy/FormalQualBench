\section{Burnside's prime-degree theorem}

\begin{theorem}[Burnside's theorem on prime degree]
\label{thm:burnside-prime-degree}
\lean{BurnsidePrimeDegreeTheorem.MainTheorem}
\notready
Let $G$ act transitively on a finite set $\alpha$ of prime cardinality. Then either the action is $2$-transitive, or $G$ has a normal regular subgroup.
\end{theorem}

\begin{proof}
The theorem statement is formalized, but its proof is currently a \texttt{sorry}. The repository also contains auxiliary Burnside-development files, but the present \texttt{MainTheorem} does not yet depend on them.
\end{proof}
