\section{Jordan's cycle theorem}

\begin{theorem}[Jordan's cycle theorem]
\label{thm:jordan-cycle}
\lean{JordanCycleTheorem.MainTheorem}
\notready
Let $G$ be a primitive finite permutation group. If $G$ contains a cycle of prime support size $p$ and $p+3\le |\alpha|$, then the alternating group on $\alpha$ is contained in $G$.
\end{theorem}
