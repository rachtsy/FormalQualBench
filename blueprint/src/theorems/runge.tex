\section{Runge's theorem}

\begin{theorem}[Runge approximation theorem]
\label{thm:runge}
\lean{RungeTheorem.MainTheorem}
\notready
Let $U\subseteq\mathbb C$ be open, $K\subseteq U$ compact with connected complement, and let $f$ be holomorphic on $U$. For every $\varepsilon>0$ there is a polynomial $p$ with $|p(z)-f(z)|<\varepsilon$ for all $z\in K$.
\end{theorem}
