\section{Borsuk--Ulam theorem}

\begin{definition}[Unit sphere]
\label{def:borsuk-ulam-sphere}
\lean{BorsukUlamTheorem.UnitSphere}
\leanok
The type $S^n$ is represented as the unit sphere in $\mathbb R^{n+1}$.
\end{definition}

\begin{definition}[Antipodal map]
\label{def:borsuk-ulam-antipode}
\lean{BorsukUlamTheorem.antipode}
\leanok
\uses{def:borsuk-ulam-sphere}
The antipodal map sends $x$ to $-x$ on the unit sphere.
\end{definition}

\begin{theorem}[Borsuk--Ulam]
\label{thm:borsuk-ulam}
\lean{BorsukUlamTheorem.MainTheorem}
\notready
\uses{def:borsuk-ulam-antipode}
Every continuous map $f:S^n\to\mathbb R^n$ takes the same value at some
antipodal pair $x$ and $-x$.
\end{theorem}
